// ============================================================ // ÁLAV ERGA — Apps Script v3 // Soporta: ventas, abonos, cuentas abiertas en tiempo real // ============================================================ function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); const tipo = data.tipo; // ── VENTA (cierre de día) ── if (tipo === 'venta') { let sheet = ss.getSheetByName('Ventas'); if (!sheet) { sheet = ss.insertSheet('Ventas'); sheet.appendRow(['Fecha','Producto','Cantidad','Venta Total','Costo Total','Utilidad','NocheID','Efectivo','Tarjeta']); } const nocheID = data.nocheID || (data.fecha + '_' + data.ts); data.items.forEach(item => { sheet.appendRow([data.fecha, item.nombre, item.qty, item.venta, item.costo, item.util, nocheID, data.efectivo||0, data.tarjeta||0]); }); } // ── ABONO ── if (tipo === 'abono') { let sheet = ss.getSheetByName('Abonos'); if (!sheet) { sheet = ss.insertSheet('Abonos'); sheet.appendRow(['Fecha','Monto','Timestamp']); } sheet.appendRow([data.fecha, data.monto, new Date().toISOString()]); } // ── CUENTA ABIERTA: crear o actualizar ── if (tipo === 'cuenta_update') { let sheet = ss.getSheetByName('CuentasAbiertas'); if (!sheet) { sheet = ss.insertSheet('CuentasAbiertas'); sheet.appendRow(['ID','Nombre','Items','Total','Pago','Estado','Timestamp']); } // Buscar si ya existe esta cuenta const lastRow = sheet.getLastRow(); let found = false; if (lastRow > 1) { const ids = sheet.getRange(2, 1, lastRow-1, 1).getValues(); for (let i = 0; i < ids.length; i++) { if (ids[i][0] === data.id) { sheet.getRange(i+2, 1, 1, 7).setValues([[ data.id, data.nombre, JSON.stringify(data.items), data.total, data.pago||'', data.estado||'abierta', new Date().toISOString() ]]); found = true; break; } } } if (!found) { sheet.appendRow([data.id, data.nombre, JSON.stringify(data.items), data.total, data.pago||'', data.estado||'abierta', new Date().toISOString()]); } } // ── CUENTA CERRADA: eliminar directamente ── if (tipo === 'cuenta_cerrar') { const sheet = ss.getSheetByName('CuentasAbiertas'); if (sheet && sheet.getLastRow() > 1) { const ids = sheet.getRange(2, 1, sheet.getLastRow()-1, 1).getValues(); for (let i = ids.length-1; i >= 0; i--) { if (ids[i][0] === data.id) { sheet.deleteRow(i+2); break; } } } } // ── LIMPIAR CUENTAS COBRADAS ── if (tipo === 'limpiar_cobradas') { const sheet = ss.getSheetByName('CuentasAbiertas'); if (sheet && sheet.getLastRow() > 1) { const estados = sheet.getRange(2, 6, sheet.getLastRow()-1, 1).getValues(); for (let i = estados.length-1; i >= 0; i--) { if (estados[i][0] === 'cobrada') { sheet.deleteRow(i+2); } } } } return ContentService .createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const tipo = e.parameter.tipo || 'dashboard'; const resultado = {}; // ── CUENTAS ABIERTAS (para POS en tiempo real) ── if (tipo === 'pos') { resultado.cuentas = []; const sheet = ss.getSheetByName('CuentasAbiertas'); if (sheet && sheet.getLastRow() > 1) { const rows = sheet.getRange(2, 1, sheet.getLastRow()-1, 7).getValues(); rows.forEach(r => { if (r[5] === 'abierta') { try { resultado.cuentas.push({ id: r[0], nombre: r[1], items: JSON.parse(r[2]||'[]'), total: r[3], pago: r[4], ts: new Date(r[6]).getTime() }); } catch(e) {} } }); } return ContentService .createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } // ── DASHBOARD ── resultado.ventas = []; resultado.abonos = []; const sheetVentas = ss.getSheetByName('Ventas'); if (sheetVentas && sheetVentas.getLastRow() > 1) { const rows = sheetVentas.getRange(2, 1, sheetVentas.getLastRow()-1, 9).getValues(); const noches = {}; rows.forEach(r => { const nocheID = r[6] || r[0]; if (!noches[nocheID]) { noches[nocheID] = { fecha: r[0], items: [], totalVenta:0, totalCosto:0, totalUtil:0, efectivo: Number(r[7])||0, tarjeta: Number(r[8])||0, ts: new Date(String(nocheID).split('_')[1]||Date.now()).getTime()||Date.now() }; } noches[nocheID].items.push({nombre:r[1],qty:r[2],venta:r[3],costo:r[4],util:r[5]}); noches[nocheID].totalVenta += Number(r[3]); noches[nocheID].totalCosto += Number(r[4]); noches[nocheID].totalUtil += Number(r[5]); }); resultado.ventas = Object.values(noches); } const sheetAbonos = ss.getSheetByName('Abonos'); if (sheetAbonos && sheetAbonos.getLastRow() > 1) { const rows = sheetAbonos.getRange(2,1,sheetAbonos.getLastRow()-1,3).getValues(); resultado.abonos = rows.map(r=>({fecha:r[0],monto:r[1],ts:new Date(r[2]).getTime()})); } return ContentService .createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ok:false,error:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
